cmake_minimum_required(VERSION 3.24)
project(plank_common_dependency_checks LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_TESTS "Exercise Host GoogleTest support as well as runtime helpers" ON)
get_filename_component(HOST_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." REALPATH)

# Exercise the production dependency wiring without capture/encoder SDKs.
include("${HOST_SOURCE_DIR}/cmake/dependencies/lizardbyte_common.cmake")
enable_testing()
add_executable(common_runtime_check runtime.cpp)
target_link_libraries(common_runtime_check PRIVATE lizardbyte::common)
add_test(NAME common_runtime COMMAND common_runtime_check)

if(BUILD_TESTS)
    get_target_property(GTEST_SOURCE_DIR gtest SOURCE_DIR)
    get_filename_component(EXPECTED_GTEST_SOURCE_DIR
            "${HOST_SOURCE_DIR}/third-party/googletest/googletest" REALPATH)
    get_filename_component(GTEST_SOURCE_DIR "${GTEST_SOURCE_DIR}" REALPATH)
    if(NOT GTEST_SOURCE_DIR STREQUAL EXPECTED_GTEST_SOURCE_DIR)
        message(FATAL_ERROR "Test support selected the nested GoogleTest source")
    endif()
    if(INSTALL_GTEST OR INSTALL_GMOCK)
        message(FATAL_ERROR "Host tests must not install GoogleTest/GoogleMock")
    endif()
    add_executable(common_test_support_check test_support.cpp)
    target_link_libraries(common_test_support_check PRIVATE lizardbyte::test_support gtest_main)
    add_test(NAME common_test_support COMMAND common_test_support_check)
else()
    foreach(TEST_TARGET gtest gtest_main gmock gmock_main lizardbyte::test_support)
        if(TARGET ${TEST_TARGET})
            message(FATAL_ERROR "BUILD_TESTS=OFF configured test target ${TEST_TARGET}")
        endif()
    endforeach()
endif()
